//--------------------------------------------------------------------------
function _onDragOver(event) {
    var targetElement, targetRect, targetIndex;
    var rootEl = this.el;
    var target = event.target;
    var options = this.options;
    var group = options.group;
    var activeSortable = eB.active;
    var sameGroup = el === group;
    var canSort = options.sort;
    var fromSortable = ec || activeSortable;
    var currentSortable = this;
    var dragEl = K;  // 当前拖拽的元素
    var cloneEl = $; // 克隆元素
    var completed = false;

    // 如果正在延迟拖拽，直接返回
    if (ex) return;

    // 阻止默认行为
    if (event.cancelable) {
        event.preventDefault();
    }

    // 获取可拖拽的目标元素
    target = y(target, options.draggable, rootEl, true);

    // 触发 dragOver 事件
    et("dragOver");

    // 如果事件被取消，返回
    if (eB.eventCanceled) return completed;

    // 如果目标在拖拽元素内部，或者是动画中的元素，返回 false
    if (K.contains(event.target) || 
        target.animated && target.animatingX && target.animatingY ||
        currentSortable._ignoreWhileAnimating === target) {
        return ea(false);
    }

    // 重置状态
    eb = false;

    // 检查是否可以放置
    if (activeSortable && !options.disabled) {
        var canInsert = sameGroup ? canSort : false;
        var canPull = ec === this || 
            (this.lastPutMode = el.checkPull(this, activeSortable, K, event)) && 
            group.checkPut(this, activeSortable, K, event);

        if (canInsert || canPull) {
            // 获取拖拽方向（垂直/水平）
            var direction = this._getDirection(event, target);
            var dragRect = N(K);  // 获取拖拽元素的位置信息

            et("dragOverValid");

            if (eB.eventCanceled) return completed;

            // 如果是相同组且可以排序
            if (canInsert) {
                Y = q;  // 设置父容器
                en();   // 清理操作
                this._hideClone();
                et("revert");

                if (!eB.eventCanceled) {
                    if (cloneEl) {
                        q.insertBefore(K, cloneEl);
                    } else {
                        q.appendChild(K);
                    }
                    return ea(true);
                }
            }

            // 查找目标位置并插入
            var nearestElement = T(rootEl, options.draggable);
            
            if (nearestElement) {
                if (nearestElement === K) return ea(false);
                
                if (nearestElement && rootEl === event.target) {
                    target = nearestElement;
                }

                if (target) {
                    targetRect = N(target);
                }

                // 执行插入逻辑
                if (eQ(q, rootEl, K, dragRect, target, targetRect, event, !!target) !== false) {
                    en();
                    if (nearestElement && nearestElement.nextSibling) {
                        rootEl.insertBefore(K, nearestElement.nextSibling);
                    } else {
                        rootEl.appendChild(K);
                    }
                    Y = rootEl;
                    ei();
                    return ea(true);
                }
            }
        }
    }
}
